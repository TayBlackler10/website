/** * M2 PT Lead System — Apps Script backend * ---------------------------------------- * Deploy this bound to your "M2 PT Leads" Google Sheet as a Web App * (Deploy > New deployment > type: Web app, execute as: Me, access: Anyone). * The resulting /exec URL goes into APPS_SCRIPT_URL in dashboard.html. * * Fields match the GymMaster "M2 Training Club | Free PT" questionnaire * (the one that currently emails Tim when a member completes it). * * See the setup guide for full step-by-step instructions. */ const SHEET_NAME = "Leads"; // Each trainer's real email address. Assigning a lead in the dashboard // (by dropdown or drag-and-drop) sends that trainer an email automatically. const TRAINER_EMAILS = { "Bekka": "bekka@m2club.co.nz", "Dave": "davidblint@gmail.com", "Madi": "madeleinecollins4@gmail.com", "Te Ao": "teaokura@gmail.com", "Rodney": "rodwonglyn@gmail.com", "Taylor": "taylor@m2club.co.nz" }; // Head coach — cc'd on assignment emails so Tim can see it went out. const HEAD_COACH_EMAIL = "tim@m2club.co.nz"; // Columns, in sheet order. Maps 1:1 to the GymMaster questionnaire fields. const COLUMNS = [ "ID", "ReceivedAt", "Name", "Phone", "Email", "ReasonForJoining", "TrainerPreference", "TrainingStyle", "Injuries", "PreferredTime", "Source", "Status", "Trainer", "AssignedAt", "Notes", "LastUpdated" ]; function getSheet_() { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) { sheet = ss.insertSheet(SHEET_NAME); } if (sheet.getLastRow() === 0) { sheet.appendRow(COLUMNS); sheet.setFrozenRows(1); } return sheet; } function colIndex_(name) { return COLUMNS.indexOf(name) + 1; // 1-based } function rowToLead_(row) { const obj = {}; COLUMNS.forEach((c, i) => { obj[c] = row[i]; }); return { id: String(obj.ID), receivedAt: obj.ReceivedAt, name: obj.Name, phone: obj.Phone, email: obj.Email, reasonForJoining: obj.ReasonForJoining || "", trainerPreference: obj.TrainerPreference || "", trainingStyle: obj.TrainingStyle || "", injuries: obj.Injuries || "", preferredTime: obj.PreferredTime || "", source: obj.Source, status: obj.Status || "New", trainer: obj.Trainer || "", assignedAt: obj.AssignedAt || "", notes: obj.Notes || "", lastUpdated: obj.LastUpdated || "" }; } function doGet(e) { const action = (e.parameter.action || "list"); if (action === "list") { const sheet = getSheet_(); const values = sheet.getDataRange().getValues(); const rows = values.slice(1).filter(r => r[0] !== ""); // skip header, skip blanks const leads = rows.map(rowToLead_); return jsonOut_({ leads }); } return jsonOut_({ error: "Unknown action" }); } function doPost(e) { let payload; try { payload = JSON.parse(e.postData.contents); } catch (err) { return jsonOut_({ error: "Bad JSON" }); } const action = payload.action; if (action === "addLead") { return jsonOut_(addLead_(payload)); } if (action === "updateLead") { return jsonOut_(updateLead_(payload)); } return jsonOut_({ error: "Unknown action" }); } function addLead_(p) { const sheet = getSheet_(); const id = Utilities.getUuid(); const now = new Date().toISOString(); const row = [ id, now, p.name || "", p.phone || "", p.email || "", p.reasonForJoining || "", p.trainerPreference || "", p.trainingStyle || "", p.injuries || "", p.preferredTime || "", p.source || "Manual", "New", "", "", "", now ]; sheet.appendRow(row); return { ok: true, id }; } function updateLead_(p) { const sheet = getSheet_(); const values = sheet.getDataRange().getValues(); const idCol = colIndex_("ID") - 1; let rowIndex = -1; for (let i = 1; i < values.length; i++) { if (String(values[i][idCol]) === String(p.id)) { rowIndex = i + 1; // 1-based sheet row break; } } if (rowIndex === -1) { return { ok: false, error: "Lead not found" }; } const prevTrainer = sheet.getRange(rowIndex, colIndex_("Trainer")).getValue(); const now = new Date().toISOString(); if (typeof p.status !== "undefined") { sheet.getRange(rowIndex, colIndex_("Status")).setValue(p.status); } if (typeof p.trainer !== "undefined") { sheet.getRange(rowIndex, colIndex_("Trainer")).setValue(p.trainer); if (p.trainer && p.trainer !== prevTrainer) { sheet.getRange(rowIndex, colIndex_("AssignedAt")).setValue(now); } } if (typeof p.notes !== "undefined") { sheet.getRange(rowIndex, colIndex_("Notes")).setValue(p.notes); } sheet.getRange(rowIndex, colIndex_("LastUpdated")).setValue(now); // Notify the trainer if they were newly assigned. if (p.trainer && p.trainer !== prevTrainer) { notifyTrainer_(sheet, rowIndex, p.trainer); } return { ok: true }; } // Link back to the hosted dashboard, included at the bottom of assignment // emails. Update this if the dashboard ever moves. const DASHBOARD_URL = "http://m2club.co.nz/M2_PT.html"; function notifyTrainer_(sheet, rowIndex, trainerName) { const email = TRAINER_EMAILS[trainerName]; if (!email) return; // no address on file — skip silently const rowVals = sheet.getRange(rowIndex, 1, 1, COLUMNS.length).getValues()[0]; const lead = rowToLead_(rowVals); const subject = "New PT lead: " + (lead.name || "Unnamed lead"); // Only include rows for fields that actually have a value — no more // "Trainer preference: -" placeholders cluttering the email. const detailRows = [ { label: "Reason for joining", value: lead.reasonForJoining }, { label: "Trainer preference", value: lead.trainerPreference }, { label: "Training style", value: lead.trainingStyle }, { label: "Preferred time", value: lead.preferredTime } ].filter(r => r.value); const receivedFormatted = formatDateNZ_(lead.receivedAt); const htmlRows = detailRows.map(r => '' + r.label + '' + escapeHtml_(r.value) + "" ).join(""); const injuryBlock = lead.injuries ? '
' + "Injury / medical note: " + escapeHtml_(lead.injuries) + "
" : ""; const html = '
' + '
' + 'M2' + 'PT LEAD SYSTEM' + "
" + '
' + '

Hi ' + escapeHtml_(trainerName) + ", you've been assigned a new lead — please follow up within 24 hours.

" + '' + '" + '" + '" + htmlRows + "
Name' + escapeHtml_(lead.name || "-") + "
Phone' + escapeHtml_(lead.phone || "-") + "
Email' + escapeHtml_(lead.email || "-") + "
" + injuryBlock + '

Source: ' + escapeHtml_(lead.source || "-") + " · Received: " + receivedFormatted + "

" + 'Open dashboard' + "
" + "
"; // Plain-text fallback for clients that don't render HTML. const textLines = [ "Hi " + trainerName + ", you've been assigned a new lead — please follow up within 24 hours.", "", "Name: " + (lead.name || "-"), "Phone: " + (lead.phone || "-"), "Email: " + (lead.email || "-") ]; detailRows.forEach(r => textLines.push(r.label + ": " + r.value)); if (lead.injuries) textLines.push("Injury / medical note: " + lead.injuries); textLines.push("", "Source: " + (lead.source || "-") + " · Received: " + receivedFormatted, "", "Dashboard: " + DASHBOARD_URL); MailApp.sendEmail({ to: email, cc: HEAD_COACH_EMAIL, subject: subject, body: textLines.join("\n"), htmlBody: html }); } function formatDateNZ_(isoString) { if (!isoString) return "-"; try { return Utilities.formatDate(new Date(isoString), "Pacific/Auckland", "d MMM, h:mm a"); } catch (e) { return isoString; } } function escapeHtml_(str) { return String(str || "") .replace(/&/g, "&") .replace(//g, ">") .replace(/"/g, """); } function jsonOut_(obj) { return ContentService.createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); } /** * ===================== GMAIL AUTO-IMPORT ===================== * Watches a dedicated Gmail inbox for GymMaster "M2 Training Club | Free PT" * questionnaire completion emails and adds each one as a New lead — no * Power Automate, no Outlook, no separate platform. * * IMPORTANT: this only works if GYMMASTER_INBOX_EMAIL below is the SAME * Google account that owns this Apps Script project and Sheet. GmailApp * always reads the inbox of whoever authorized the script — it can't reach * into an unrelated mailbox. If your dedicated Gmail is a different account * to the one that owns the Sheet, this needs to be redeployed from that * account instead. * * IMPORTANT #2: the actual answers aren't in the email body — GymMaster's * notification email is just a one-line "so-and-so completed the * questionnaire" message with a PDF attached. This script reads that PDF * (via Drive, which extracts its text) rather than the email itself, which * means the "Drive API" advanced service must be enabled first: * in the Apps Script editor, click Services (+) in the left sidebar, * find "Drive API", and click Add. * * Setup, in order: * 1. Add the Drive API advanced service as described above. * 2. In GymMaster, add GYMMASTER_INBOX_EMAIL as a second Email Recipient on * the "M2 Training Club | Free PT" questionnaire, alongside Tim. * 3. Wait for one real completed questionnaire to land in that inbox. * 4. Open this script, select debugLatestGymMasterEmail in the function * dropdown at the top, click Run, then check View > Logs. Confirm each * parsed field looks right. If something's blank or wrong, the question * wording in QUESTION_LABELS below doesn't match the real PDF text — fix * the label text (copy it exactly from the logged PDF text) and run again. * 5. Once the parsed output looks correct, select createGmailImportTrigger * in the function dropdown and click Run ONCE. That turns on an * automatic check every 10 minutes — nothing else to do after that. * =============================================================== */ // The inbox that receives GymMaster's questionnaire completion emails. // This is the same Google account that owns this script and the Sheet, // so GmailApp below can actually read it. const GYMMASTER_INBOX_EMAIL = "tayblacklerfitness@gmail.com"; // Gmail search used to find completion emails, excluding ones already // imported. GymMaster's subject line is just "Questionnaire Updated" (not // the questionnaire name), so we match on the sender name plus "Free PT" // appearing anywhere in the email instead. const GYMMASTER_SEARCH_QUERY = 'from:"M2 Training Club" "Free PT" -label:m2-imported'; // Label applied to a thread once it's been turned into a lead, so the same // email is never imported twice. const IMPORTED_LABEL = "m2-imported"; // The three dropdown questions on the form. GymMaster's PDF only ever prints // the SELECTED option's text (never the other choices), and — for reasons // specific to how it lays out dropdown widgets — that text can land in an // unpredictable spot relative to its own question label. So instead of // looking for "the answer right after this label", we just search the whole // PDF text for one of these known option strings; whichever one appears IS // the answer, regardless of where it landed. const DROPDOWN_OPTIONS = { trainerPreference: ["a Male trainer", "a Female trainer", "I do not have a preference"], trainingStyle: [ "Strength Training - weights & machines", "Olympic Lifting", "High Intensity Interval or Circuit Training", "Cardio/Fitness Focus", "Mobility/Movement", "Bodyweight", "Power Lifting", "Sports Specific", "Not sure" ], preferredTime: [ "Weekday - Early morning", "Weekday - Midday", "Weekday - Evenings", "Weekend - Morning / Midday", "Any - I'm flexible" ] }; // Every question, in the order it appears on the form — including the // dropdowns, even though their answers are pulled separately via // DROPDOWN_OPTIONS above. Text-type answers are captured as "everything // between this label and the next one in this list", so the dropdown // entries still need to be here just to mark where a text answer should // stop reading. const ALL_LABELS_IN_ORDER = [ { field: "name", label: "Your First & Last Name", type: "text" }, { field: "reasonForJoining", label: "What best describes your reason for joining the gym?", type: "text" }, { field: "trainerPreference", label: "Your new membership includes a free Personal Training session with one of our trainers. What would be your preference:", type: "dropdown" }, { field: "trainingStyle", label: "What best describes your preferred training style?", type: "dropdown" }, { field: "injuries", label: "Do you have any injuries or medical issues our trainers should be aware of?", type: "text" }, { field: "preferredTime", label: "What time of the day would suit you best for your free PT session?", type: "dropdown" }, { field: "phone", label: "Phone Number to contact on", type: "text" } ]; function importGymMasterLeads_() { const threads = GmailApp.search(GYMMASTER_SEARCH_QUERY, 0, 20); if (threads.length === 0) return; const label = getOrCreateLabel_(IMPORTED_LABEL); threads.forEach(thread => { thread.getMessages().forEach(msg => { const pdf = getPdfAttachment_(msg); if (!pdf) return; // no PDF on this message — nothing to parse, skip it const text = pdfToText_(pdf); const parsed = parseGymMasterEmail_(text); if (!parsed.email) parsed.email = extractSenderEmail_(msg); parsed.source = "GymMaster"; addLead_(parsed); }); thread.addLabel(label); }); } // GymMaster's notification email is just a link — the actual answers are in // the attached PDF. This finds it. function getPdfAttachment_(msg) { const attachments = msg.getAttachments({ includeAttachments: true }); return attachments.find(a => a.getContentType() === "application/pdf") || null; } // Converts a PDF blob to plain text by round-tripping it through Google // Drive (Drive auto-extracts embedded text, or OCRs it if it's a scanned // image). Requires the "Drive API" advanced service — see setup guide. // Handles both the v2 (Drive.Files.insert) and v3 (Drive.Files.create) // versions of that service, since the default depends on how it was added. function pdfToText_(blob) { const metadata = { name: blob.getName(), title: blob.getName(), mimeType: MimeType.GOOGLE_DOCS }; const file = Drive.Files.create ? Drive.Files.create(metadata, blob, { ocr: true }) // v3 : Drive.Files.insert(metadata, blob, { convert: true, ocr: true }); // v2 const fileId = file.id; try { const doc = DocumentApp.openById(fileId); return doc.getBody().getText(); } finally { try { Drive.Files.remove(fileId); // clean up the temporary converted copy } catch (e) { Logger.log("Couldn't delete temp Drive file " + fileId + " (harmless): " + e.message); } } } function parseGymMasterEmail_(rawText) { // Collapse all whitespace/line-wrapping down to single spaces first, so a // label that happens to wrap across two PDF lines still matches as one // continuous string. const text = rawText.replace(/\s+/g, " ").trim(); const result = {}; // Dropdowns: find whichever known option string is present. result.trainerPreference = findKnownOption_(text, DROPDOWN_OPTIONS.trainerPreference); result.trainingStyle = findKnownOption_(text, DROPDOWN_OPTIONS.trainingStyle); result.preferredTime = findKnownOption_(text, DROPDOWN_OPTIONS.preferredTime); // Typed fields: grab everything between this label and the next one in // the full question order (skipping the dropdown entries themselves, // since those are already answered above — they're only here as boundaries). for (let i = 0; i < ALL_LABELS_IN_ORDER.length; i++) { const item = ALL_LABELS_IN_ORDER[i]; if (item.type !== "text") continue; const nextLabel = i + 1 < ALL_LABELS_IN_ORDER.length ? ALL_LABELS_IN_ORDER[i + 1].label : null; result[item.field] = extractAnswer_(text, item.label, nextLabel); } // The Phone field is last in the form, so its captured text can have the // deferred "preferred time" dropdown value stuck on the end of it (the // same PDF quirk mentioned above) — trim that back off if present. if (result.phone) { DROPDOWN_OPTIONS.preferredTime.forEach(opt => { const idx = result.phone.indexOf(opt); if (idx !== -1) result.phone = result.phone.substring(0, idx).trim(); }); } return result; } function findKnownOption_(text, options) { for (const opt of options) { if (text.indexOf(opt) !== -1) return opt; } return ""; } function extractAnswer_(text, label, nextLabel) { const startIdx = text.indexOf(label); if (startIdx === -1) return ""; const afterLabel = startIdx + label.length; let endIdx = nextLabel ? text.indexOf(nextLabel, afterLabel) : -1; if (endIdx === -1) endIdx = Math.min(text.length, afterLabel + 300); // last field: grab a bounded chunk return text.substring(afterLabel, endIdx).trim(); } function extractSenderEmail_(msg) { const from = msg.getFrom(); // e.g. "John Smith " const match = from.match(/<(.+)>/); return match ? match[1] : from; } function getOrCreateLabel_(name) { return GmailApp.getUserLabelByName(name) || GmailApp.createLabel(name); } // Run manually (no trigger) after a real completion email has arrived, to // sanity-check the parsing before turning on automatic imports. function debugLatestGymMasterEmail() { const threads = GmailApp.search('from:"M2 Training Club" "Free PT"', 0, 1); if (threads.length === 0) { Logger.log("No matching emails found yet. Check GYMMASTER_SEARCH_QUERY matches the real sender/subject."); return; } const messages = threads[0].getMessages(); const msg = messages[messages.length - 1]; const pdf = getPdfAttachment_(msg); if (!pdf) { Logger.log("No PDF attachment found on this email. Raw body was:"); Logger.log(msg.getPlainBody()); return; } const text = pdfToText_(pdf); Logger.log("----- TEXT EXTRACTED FROM PDF -----"); Logger.log(text); Logger.log("----- PARSED FIELDS -----"); Logger.log(JSON.stringify(parseGymMasterEmail_(text), null, 2)); // Also write it straight into the Sheet — easier to read/copy than the // Apps Script log viewer. Check the "Debug" tab after running this. const ss = SpreadsheetApp.getActiveSpreadsheet(); let debugSheet = ss.getSheetByName("Debug") || ss.insertSheet("Debug"); debugSheet.clear(); debugSheet.getRange(1, 1).setValue("Raw PDF text:"); debugSheet.getRange(2, 1).setValue(text); debugSheet.getRange(4, 1).setValue("Parsed fields:"); debugSheet.getRange(5, 1).setValue(JSON.stringify(parseGymMasterEmail_(text), null, 2)); } // Run ONCE to turn on automatic checking every 10 minutes. Safe to re-run — // it clears any existing trigger for this function first so you never end // up with duplicates. function createGmailImportTrigger() { ScriptApp.getProjectTriggers().forEach(t => { if (t.getHandlerFunction() === "importGymMasterLeads_") ScriptApp.deleteTrigger(t); }); ScriptApp.newTrigger("importGymMasterLeads_") .timeBased() .everyMinutes(10) .create(); Logger.log("Trigger created — checking for new GymMaster leads every 10 minutes."); }
